/* ============================================================
   LIVE CLOCK (12-HOUR FORMAT WITH AM/PM)
============================================================ */
function updateClock() {
  const now = new Date();
  let hours = now.getHours();
  const minutes = String(now.getMinutes()).padStart(2, '0');
  const seconds = String(now.getSeconds()).padStart(2, '0');
  const ampm = hours >= 12 ? 'PM' : 'AM';
  hours = hours % 12;
  hours = hours ? hours : 12;
  const hoursStr = String(hours).padStart(2, '0');
  const clockEl = document.getElementById('liveClock');
  const dateEl  = document.getElementById('liveDate');
  if (clockEl) {
    clockEl.textContent = `${hoursStr}:${minutes}:${seconds} ${ampm}`;
  }
  if (dateEl) {
    const days   = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
    const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    dateEl.textContent = `${days[now.getDay()]}, ${months[now.getMonth()]} ${now.getDate()}, ${now.getFullYear()}`;
  }
}
setInterval(updateClock, 1000);
updateClock();

function handleSpecialCommand(cmd) {

  if (c === '/clear') return { isCommand: true, response: "🗑️ **Chat Cleared** — Conversation history has been reset. Fresh start! What would you like to explore?" };
  if (c === '/help') return {
    isCommand: true,
    response: `🤖 **AURA Commands**\n\n` +
      `\`/help\` — Show this help menu\n` +
      `\`/clear\` — Clear the conversation\n` +
      `\`/focus\` — Start a focus session tip\n` +
      `\`/idea\` — Trigger idea generation mode\n` +
      `\`/decide\` — Open decision framework\n` +
      `\`/scan\` — Risk scanning guidance\n\n` +
      `Or just ask me anything naturally! I understand questions about study, career, emotions, finance, habits, creativity, technology, relationships, and much more. 🌟`
  };
  if (c === '/focus') return { isCommand: true, response: "🎯 **Focus Mode Activated!**\n\nHere's your AURA focus protocol:\n\n1. **Clear your space** — Remove all non-essential items from your desk\n2. **Phone away** — Put it face-down in another room if possible\n3. **Set your intention** — Write exactly one sentence about what you'll accomplish\n4. **Start the timer** — 25 minutes, no exceptions\n5. **Breathe first** — 3 deep breaths before beginning\n\nRemember: the goal is not to feel motivated, it's to start. /* ============================================================
   KEYWORD PATTERN MATCHING
============================================================ */
function detectCategory(msg) {
  const m = msg.toLowerCase();

  if (/\b(hello|hi|hey|howdy|greetings|good morning|good afternoon|good evening|sup|what'?s up)\b/.test(m)) return 'greetings';
  if (/\b(bye|goodbye|farewell|see you|later|take care|goodnight|signing off|until next time)\b/.test(m)) return 'farewells';
  if (/\b(study|learn|memorize|flashcard|exam|test|quiz|retention|school|college|university|academic|education|course|lecture|reading|textbook|notes|revision)\b/.test(m)) return 'study';
  if (/\b(focus|concentrate|distract|productivity|procrastinat|pomodoro|deep work|flow state|attention|task|work session|efficient)\b/.test(m)) return 'focus';
  if (/\b(decid|choice|option|dilemma|choose|stuck|uncertain|weighing|pros and cons|should i|what should|help me decide)\b/.test(m)) return 'decisions';
  if (/\b(feel|emotion|sad|anxious|depress|overwhelm|lonely|angry|frustrated|mental health|therapy|mood|emotional|stress|burnout|grief|loss|scared|worried|fear)\b/.test(m)) return 'emotions';
  if (/\b(career|job|work|profession|salary|promotion|resume|interview|skill|industry|employ|hire|boss|manager|colleague|office|workplace|profession)\b/.test(m)) return 'career';
  if (/\b(habit|routine|daily|discipline|consistent|schedule|morning|ritual|practice|streak|track)\b/.test(m)) return 'habits';
  if (/\b(money|finance|budget|invest|saving|debt|income|expense|wealth|financial|stock|crypto|retirement|fund|insurance|tax|loan|mortgage|credit)\b/.test(m)) return 'finance';
  if (/\b(idea|creative|brainstorm|invent|innovate|concept|startup|product|solution|original|inspiration|imagine|design)\b/.test(m)) return 'ideas';
  if (/\b(risk|danger|scam|fraud|threat|security|hack|vulnerable|warning|red flag|unsafe|phishing|identity theft|protect)\b/.test(m)) return 'risk';
  if (/\b(health|diet|exercise|sleep|fitness|workout|nutrition|wellness|medical|body|weight|run|gym|yoga|meditat|immune|supplement)\b/.test(m)) return 'health';
  if (/\b(motivat|inspire|give up|quit|keep going|persevere|mindset|believe|confiden|self.doubt|push through|success|goal|dream|achieve)\b/.test(m)) return 'motivation';
  if (/\b(ai|artificial intelligence|machine learning|tech|software|coding|programming|blockchain|crypto|app|algorithm|data|cloud|automation|robot|neural)\b/.test(m)) return 'technology';
  if (/\b(time|schedule|calendar|priorit|urgent|deadline|manage|organize|plan|efficient|overwhelm with tasks|too much|agenda|balance)\b/.test(m)) return 'time_management';
  if (/\b(relationship|friend|partner|family|love|communication|conflict|trust|bond|connect|lonely|marriage|dating|social|colleague|team)\b/.test(m)) return 'relationships';
  if (/\b(self.improve|grow|better version|potential|self.aware|mindful|reflect|journal|wisdom|knowledge|develop|transform)\b/.test(m)) return 'self_improvement';
  if (/\b(stress|anxiet|overwhelm|panic|worry|nervous|tension|pressure|burnout|exhaust|mental load|cope)\b/.test(m)) return 'stress';
  if (/\b(aura|who are you|what are you|your purpose|about you|what can you do|capabilities|your name|are you ai|are you human)\b/.test(m)) return 'aura_self';

  return 'fallback';
}

/* ============================================================
   MAIN AI RESPONSE FUNCTION
============================================================ */
function getAIResponse(message, history = []) {
  const cleaned = message.trim();
  const lower   = cleaned.toLowerCase();

  // Special commands
  if (cleaned.startsWith('/')) {
    const result = handleSpecialCommand(cleaned.split(' ')[0]);
    if (result.isCommand) {
      if (cleaned.startsWith('/clear')) {
        clearChat();
        return null; // null means we handled it specially
      }
      return result.response;
    }
  }

  // Category detection
  const category = detectCategory(lower);
  const pool = aiResponses[category] || aiResponses.fallback;

  // Rotate response based on message length to add variety
  const idx = cleaned.length % pool.length;
  return pool[idx];
}
